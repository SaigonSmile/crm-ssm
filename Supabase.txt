1. QUAN HỆ GIỮA CÁC BẢNG (RELATIONSHIP MAP)
auth.users: Bảng gốc của Supabase quản lý thông tin đăng nhập.
profiles: Liên kết 1-1 với auth.users. Lưu trữ vai trò (role) và thông tin cá nhân.
teams &amp; team_members: Quản lý cấu trúc phòng ban. Một Trưởng nhóm quản lý một Team.
leads: Bảng trung tâm, gắn với profiles (người tạo/người phụ trách) và pipeline_stages.
tasks: Liên kết với leads và profiles (người thực hiện).
activity_logs: Ghi lại mọi biến động dữ liệu để phục vụ hậu kiểm.
2. SQL SCHEMA HOÀN CHỈNH
sql
-- 1. ENUMS &amp; EXTENSIONS
create type user_role as enum ('admin', 'team_lead', 'sales');
create type task_priority as enum ('low', 'medium', 'high');
create type task_status as enum ('todo', 'in_progress', 'done');
-- 2. TABLES
-- Profiles: Thông tin hồ sơ người dùng
create table public.profiles (
  id uuid references auth.users on delete cascade primary key,
  email text unique not null,
  full_name text,
  avatar_url text,
  role user_role default 'sales' not null,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  updated_at timestamp with time zone default timezone('utc'::text, now()) not null
);
-- Teams: Quản lý nhóm/chi nhánh
create table public.teams (
  id uuid default gen_random_uuid() primary key,
  name text not null,
  manager_id uuid references public.profiles(id),
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);
-- Team Members: Thành viên trong nhóm
create table public.team_members (
  id uuid default gen_random_uuid() primary key,
  team_id uuid references public.teams(id) on delete cascade,
  user_id uuid references public.profiles(id) on delete cascade,
  unique(team_id, user_id)
);
-- Pipeline Stages: Các bước trong phễu bán hàng (Kanban)
create table public.pipeline_stages (
  id uuid default gen_random_uuid() primary key,
  name text not null,
  position int not null, -- Thứ tự hiển thị 1, 2, 3...
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);
-- Leads: Khách hàng tiềm năng
create table public.leads (
  id uuid default gen_random_uuid() primary key,
  full_name text not null,
  email text,
  phone text not null,
  company text,
  segment text, -- VIP, Khách mới, v.v.
  interest text[], -- Mảng các sản phẩm quan tâm
  expected_value numeric(15, 2) default 0,
  stage_id uuid references public.pipeline_stages(id),
  assigned_to uuid references public.profiles(id),
  created_by uuid references public.profiles(id) default auth.uid(),
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  updated_at timestamp with time zone default timezone('utc'::text, now()) not null
);
-- Lead Tags: Nhãn phân loại
create table public.lead_tags (
  id uuid default gen_random_uuid() primary key,
  lead_id uuid references public.leads(id) on delete cascade,
  tag_name text not null
);
-- Tasks: Công việc cần làm
create table public.tasks (
  id uuid default gen_random_uuid() primary key,
  lead_id uuid references public.leads(id) on delete cascade,
  title text not null,
  description text,
  due_date timestamp with time zone,
  priority task_priority default 'medium',
  status task_status default 'todo',
  assigned_to uuid references public.profiles(id),
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);
-- Payments (Subscriptions): Giao dịch
create table public.payments (
  id uuid default gen_random_uuid() primary key,
  lead_id uuid references public.leads(id),
  amount numeric(15, 2) not null,
  status text, -- completed, pending
  payment_date timestamp with time zone default now()
);
-- Activity Logs: Nhật ký hoạt động
create table public.activity_logs (
  id uuid default gen_random_uuid() primary key,
  user_id uuid references public.profiles(id),
  action text not null, -- 'CREATE_LEAD', 'UPDATE_STAGE', v.v.
  target_id uuid,
  target_type text,
  created_at timestamp with time zone default now()
);
-- 3. INDEXES (Tối ưu hóa truy vấn)
create index idx_leads_assigned_to on public.leads(assigned_to);
create index idx_leads_stage on public.leads(stage_id);
create index idx_tasks_assigned_to on public.tasks(assigned_to);
create index idx_tasks_due_date on public.tasks(due_date);
create index idx_team_members_user on public.team_members(user_id);
-- 4. TỰ ĐỘNG TẠO PROFILE (Trigger &amp; Function)
-- Hàm này chạy mỗi khi có user mới trong auth.users
create or replace function public.handle_new_user()
returns trigger as $$
begin
  insert into public.profiles (id, email, full_name, avatar_url, role)
  values (
    new.id,
    new.email,
    new.raw_user_meta_data-&gt;&gt;'full_name',
    new.raw_user_meta_data-&gt;&gt;'avatar_url',
    'sales' -- Mặc định là Sales
  )
  on conflict (id) do update
  set email = excluded.email,
      full_name = coalesce(public.profiles.full_name, excluded.full_name),
      avatar_url = coalesce(public.profiles.avatar_url, excluded.avatar_url);
  return new;
end;
$$ language plpgsql security definer;
create trigger on_auth_user_created
  after insert on auth.users
  for each row execute procedure public.handle_new_user();
3. CHÍNH SÁCH BẢO MẬT (ROW LEVEL SECURITY - RLS)
sql
-- Kích hoạt RLS
alter table public.profiles enable row level security;
alter table public.leads enable row level security;
alter table public.tasks enable row level security;
alter table public.teams enable row level security;
-- HÀM HỖ TRỢ PHÂN QUYỀN
create or replace function is_admin() returns boolean as $$
  select exists (select 1 from public.profiles where id = auth.uid() and role = 'admin');
$$ language sql security definer;
create or replace function is_team_lead_of_user(target_user_id uuid) returns boolean as $$
  select exists (
    select 1 from public.teams t
    join public.team_members tm on t.id = tm.team_id
    where t.manager_id = auth.uid() and tm.user_id = target_user_id
  );
$$ language sql security definer;
-- POLICIES CHO BẢNG LEADS
-- 1. Admin: Xem tất cả
create policy "Admins can do everything on leads"
  on public.leads for all
  using (is_admin());
-- 2. Sales: Chỉ xem lead mình được giao hoặc tự tạo
create policy "Sales can see their own leads"
  on public.leads for select
  using (assigned_to = auth.uid() or created_by = auth.uid());
-- 3. Team Lead: Xem lead của thành viên trong nhóm
create policy "Team leads can see team leads"
  on public.leads for select
  using (is_team_lead_of_user(assigned_to));
4. QUY TRÌNH XÁC THỰC VÀ LIÊN KẾT TÀI KHOẢN (FLOWS)
4.1. Đăng ký/Đăng nhập Email + Mật khẩu
Người dùng điền Email + Pass -&gt; Gọi supabase.auth.signUp.
Supabase gửi email xác nhận. Sau khi xác nhận, Trigger on_auth_user_created tự động tạo 1 bản ghi trong bảng profiles.
Lần sau đăng nhập, hệ thống kiểm tra auth.users và cho phép vào CRM.
4.2. Đăng nhập Google OAuth
Gọi supabase.auth.signInWithOAuth({ provider: 'google' }).
Google trả về Email và Identity.
4.3. Logic Chống Tạo Trùng (Identity Linking)
Cơ chế: Supabase Auth mặc định sẽ liên kết các Identity (Google, Email/Pass) nếu chúng có cùng địa chỉ Email (với điều kiện email đã được xác nhận).
Hành động: Khi đăng nhập Google, nếu email đó đã có tài khoản Email/Pass, Supabase sẽ tự động nhận diện đó là User ID hiện tại.
Đảm bảo Profile: Trigger của chúng ta có câu lệnh on conflict (id) do update. Điều này đảm bảo dù đăng nhập bằng cách nào, user_id vẫn giữ nguyên và chỉ có duy nhất một hồ sơ trong bảng profiles.
5. CƠ CHẾ PHÂN QUYỀN THỰC TẾ
Admin: Kiểm soát bảng profiles để thay đổi Role của nhân viên. Khi Role đổi sang admin, chính sách RLS is_admin() sẽ ngay lập tức mở toàn bộ dữ liệu.
Trưởng nhóm: Phải được gán là manager_id trong bảng teams và các thành viên phải nằm trong bảng team_members tương ứng. RLS sẽ tự động tính toán quan hệ này để hiển thị dữ liệu.
Sales: Mọi dữ liệu tạo ra đều mặc định gắn created_by = auth.uid().
